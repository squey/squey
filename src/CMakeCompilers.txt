# The kernels that are built a second time for a wider instruction set only make sense
# on x86, where -mavx2 and friends exist : elsewhere the compiler rejects those flags
# outright. ARCH_TYPE is independent, as it may legitimately be emptied on x86 too.
if("${CMAKE_SYSTEM_PROCESSOR}" STREQUAL "x86_64")
    set(SQUEY_TARGET_X86_64 ON)
endif()

# ARCH_TYPE is the baseline the whole tree is built against : the submodules read it
# rather than picking their own default, and dependencies with a hand written baseline
# (google-snappy.bst) are expected to match it. Override it with -DARCH_TYPE=x86-64-v3
# for a build targeting a known machine, or set it empty to leave -march out entirely.
if(SQUEY_TARGET_X86_64 AND NOT DEFINED ARCH_TYPE)
    # x86-64-v3 requires AVX2/BMI2/FMA, introduced by Haswell in 2013: binaries built
    # against it die with SIGILL on the very first vector instruction on any older CPU,
    # before reaching main(). v2 only asks for SSE4.2/POPCNT (Nehalem, 2009), and simde
    # emulates the wider intrinsics the code uses.
    set(ARCH_TYPE "x86-64-v2")
endif()
if(ARCH_TYPE)
    set(CMAKE_CXX_ARCH_FLAGS "-march=${ARCH_TYPE}")
endif()

if (WIN32)
    set(CMAKE_CXX_COMMON_FLAGS "-Wa,-mbig-obj -Wl,--export-all-symbols -Wl,--enable-auto-import -DWIN32_LEAN_AND_MEAN")
    add_definitions(-DUNICODE -D_UNICODE)
endif()
# Passing a 256 bits simde vector by value uses a different calling convention when AVX
# is off, which -Wpsabi reports. Those types never cross a library boundary here: they
# only appear in inline static helpers and local variables, so the warning has no bearing.
set(CMAKE_CXX_COMMON_FLAGS "${CMAKE_CXX_COMMON_FLAGS} -Wall -Wextra -fPIC -Wno-psabi")

    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} --coverage")
    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} --coverage")
    message(STATUS "Code coverage instrumentation enabled")
endif()

if (ADDRESS_SANITIZER)
    # Everything below is attached to the C++ language alone rather than to the flag
    # variables, which are shared by every language:
    #  - only the C++ compiler is selected by the build, the C one stays whatever
    #    /usr/bin/cc is, that is gcc. gcc has its own libasan, so a C target linked
    #    through it pulls in a second runtime and the binaries die on startup with
    #    "Your application is linked against incompatible ASan runtimes".
    #  - -resource-dir is clang-only, and gcc stops on it as an unrecognized option.
    #    In the linker flags it broke every try_compile of the configure step, and Qt6
    #    was then reported missing because its Threads check could no longer link.
    # CMAKE_CXX_FLAGS is no safer than the linker flags here: the furl targets copy it
    # into their COMPILE_FLAGS even though their sources are C.
    # Leaving furl uninstrumented is not a loss worth compiling it with a mismatched
    # runtime for: the sanitizer still watches every allocation it makes, since the
    # executables it is linked into carry the runtime themselves.
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-fsanitize=address;-fno-omit-frame-pointer>")
    add_link_options("$<$<LINK_LANGUAGE:CXX>:-fsanitize=address;-fno-omit-frame-pointer>")
    # clang looks up its runtime libraries under -print-resource-dir, but the sysroot
    # installs them under the install prefix instead, which leaves the link step unable
    # to find libclang_rt.asan. Point at them explicitly when they are missing.
    if (CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-resource-dir
                        OUTPUT_VARIABLE CLANG_RESOURCE_DIR
                        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        file(GLOB ASAN_RUNTIME "${CLANG_RESOURCE_DIR}/lib/*/libclang_rt.asan*")
        if (NOT ASAN_RUNTIME)
            file(GLOB ASAN_RESOURCE_DIRS "${CMAKE_INSTALL_PREFIX}/lib/clang/*")
            if (ASAN_RESOURCE_DIRS)
                list(GET ASAN_RESOURCE_DIRS 0 ASAN_RESOURCE_DIR)
                add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-resource-dir=${ASAN_RESOURCE_DIR}>")
                add_link_options("$<$<LINK_LANGUAGE:CXX>:-resource-dir=${ASAN_RESOURCE_DIR}>")
            endif()
        endif()
    endif()
    message(STATUS "Address sanitizer instrumentation enabled")
endif()

set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_COMMON_FLAGS} ${CODE_COVERAGE_FLAGS}")

# Configure linker

# Use ld.mold if available
OPTION(USE_LD_MOLD "Use mold linker" ON)
if (USE_LD_MOLD)
    execute_process(COMMAND ${CMAKE_CXX_COMPILER} -fuse-ld=mold -Wl,--separate-debug-file,--version ERROR_QUIET OUTPUT_VARIABLE LD_VERSION)
    if ("${LD_VERSION}" MATCHES "mold")
        message(STATUS "using mold linker as it is available.")
        set(CMAKE_LINKER "mold")
        # --separate-debug-file leaves the debug file with a .text holding no
        # content, so the hash Breakpad falls back on to identify a module has
        # nothing to read and dump_syms gives up with "unable to generate file
        # identifier". A build id spares it that fallback, and ties the symbols
        # to the binary the minidump names. The gcc driver asks for one already,
        # clang does not, which is why the symbols of the released builds were
        # empty.
        set(MOLD_LINKER_FLAGS "-fuse-ld=mold -Wl,--build-id -Wl,--separate-debug-file")
        set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} ${MOLD_LINKER_FLAGS}")
        set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} ${MOLD_LINKER_FLAGS}")
    else ()
        message(STATUS "mold linker isn't available, using the default system linker.")
        set(USE_LD_MOLD OFF)
    endif ()
endif ()

    endif()
endif()
